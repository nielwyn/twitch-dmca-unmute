"use strict";

/*
 * Twitch DMCA Unmute
 *
 * Two separate things make a VOD look censored:
 *
 * 1. AUDIO. Twitch renames the affected segments in the media playlist
 *      1337.ts  ->  1337-muted.ts   (silence)
 *    The original stays on the CDN for ~24h, then 403s forever. We rewrite the
 *    playlist back to the originals while they still exist.
 *
 * 2. THE RED BAR. The markers on the seek bar come from GraphQL metadata
 *    (mutedSegmentConnection / muted_segments), not from the audio. They stay
 *    red even with the audio restored, so we strip them from the API response.
 */

const DEFAULTS = { enabled: true, optimistic: true, hideMarkers: true, crossQuality: false };
const settings = Object.assign({}, DEFAULTS);

const IS_PLAYLIST = /\.m3u8(\?|$|\/)/i;
const IS_SEGMENT = /\.(ts|mp4|m4s|m4v|aac)(\?|$)/i;
const MUTED_SEGMENT = /^(.*?)-muted(\.(?:ts|mp4|m4s|m4v|aac)(?:\?.*)?)$/;
const MUTED_PLAYLIST_NAME = /-muted-\w+(?=\.m3u8)/;
const AIR_DATE = /#ID3-EQUIV-TDTG:(.+)/;
const MUTED_KEY = /^muted_?segment/i;

// Hosts that plausibly serve Twitch VOD manifests. Used only to decide whether
// an extension-less URL is worth sniffing for "#EXTM3U".
const VOD_HOST = /(^|\.)(cloudfront\.net|ttvnw\.net|twitch\.tv|akamaized\.net|llnwd\.net|fastly\.net|edgecastcdn\.net)$/i;
const GQL_HOST = /(^|\.)twitch\.tv$/i;

const PROBE_TIMEOUT_MS = 4000;
const PROBE_DEADLINE_MS = 5000;   // how long a playlist may be held while probing
const PROBE_CONCURRENCY = 16;
const PROBE_BUDGET = 800;         // max segments probed per playlist
const MAX_TRACKED_SEGMENTS = 20000;
const DIAG_MAX = 60;

browser.storage.local.get(DEFAULTS).then((stored) => Object.assign(settings, stored));
browser.storage.onChanged.addListener((changes, area) => {
  if (area !== "local") return;
  for (const [key, { newValue }] of Object.entries(changes)) {
    if (key in settings) settings[key] = newValue;
  }
});

/** Absolute unmuted segment URL -> absolute muted URL, for the 403 fallback. */
const rewritten = new Map();
/** Absolute segment URL -> Promise<boolean|null>. null = unknown. */
const probes = new Map();
/** tabId -> report shown in the popup. */
const tabReports = new Map();
/** Rewritten segment URL -> which muted range it belongs to (for live failure stats). */
const segmentRange = new Map();
/** vodKey -> quality directory names seen in the master playlist. */
const qualities = new Map();
/** Ring buffer of interesting requests, for the popup's diagnostics panel. */
const diag = [];
const counters = { playlists: 0, gql: 0, sniffed: 0, muted: 0, crossQuality: 0, fallbacks: 0 };

/* ------------------------------------------------------------------ utils */

function log(action, url) {
  let host = url;
  let path = "";
  try {
    const u = new URL(url);
    host = u.host;
    path = u.pathname;
  } catch (_) {}
  diag.push({ t: Date.now(), action, host, path: path.length > 70 ? "…" + path.slice(-70) : path });
  if (diag.length > DIAG_MAX) diag.shift();
}

function formatTime(seconds) {
  const s = Math.max(0, Math.floor(seconds));
  const h = Math.floor(s / 3600);
  const m = Math.floor((s % 3600) / 60);
  const pad = (n) => String(n).padStart(2, "0");
  return h > 0 ? `${h}:${pad(m)}:${pad(s % 60)}` : `${m}:${pad(s % 60)}`;
}

function emptyReport() {
  return {
    key: null,
    ranges: [],
    apiRanges: [],
    restoredRanges: 0,
    partialRanges: 0,
    restoredSegments: 0,
    mutedSegments: 0,
    failedSegments: 0,
    ageHours: null,
    markersHidden: 0,
    note: null,
    sawPlaylist: false
  };
}

function reportFor(tabId) {
  let report = tabReports.get(tabId);
  if (!report) {
    report = emptyReport();
    tabReports.set(tabId, report);
  }
  return report;
}

function setBadge(tabId, report) {
  if (tabId < 0) return;
  let text = "";
  let color = "#00a862";
  if (report.restoredRanges > 0 || report.partialRanges > 0) {
    text = String(report.restoredRanges + report.partialRanges);
    if (report.partialRanges > 0) color = "#c78b16";
  } else if (report.ranges.length > 0 || report.apiRanges.length > 0) {
    text = "✕";
    color = "#d14343";
  } else if (report.markersHidden > 0) {
    text = "✓";
  }
  browser.browserAction.setBadgeText({ tabId, text });
  browser.browserAction.setBadgeBackgroundColor({ tabId, color });
}

function vodKeyFor(url) {
  try {
    const u = new URL(url);
    return u.origin + u.pathname.split("/").slice(0, -2).join("/");
  } catch (_) {
    return url;
  }
}

/** Same segment number under the VOD's other quality renditions, best first. */
function altSegmentUrls(absUrl) {
  try {
    const u = new URL(absUrl);
    const parts = u.pathname.split("/");
    const qIndex = parts.length - 2;
    const key = u.origin + parts.slice(0, qIndex).join("/");
    const list = qualities.get(key);
    if (!list) return [];
    return list
      .filter((q) => q !== parts[qIndex])
      .map((q) => {
        const copy = parts.slice();
        copy[qIndex] = q;
        return u.origin + copy.join("/") + u.search;
      });
  } catch (_) {
    return [];
  }
}

/** Probe many URLs with a worker pool and a hard deadline (unfinished -> null). */
async function probeMany(urls, deadlineAt) {
  const results = new Map();
  let next = 0;
  async function worker() {
    while (next < urls.length) {
      const url = urls[next++];
      if (results.has(url)) continue;
      if (Date.now() > deadlineAt && !probes.has(url)) {
        results.set(url, null);
        continue;
      }
      results.set(url, await probe(url));
    }
  }
  const workers = [];
  for (let i = 0; i < Math.min(PROBE_CONCURRENCY, urls.length); i++) workers.push(worker());
  await Promise.all(workers);
  return results;
}

function track(unmutedUrl, mutedUrl) {
  if (rewritten.size >= MAX_TRACKED_SEGMENTS) {
    rewritten.clear();
    segmentRange.clear();
  }
  rewritten.set(unmutedUrl, mutedUrl);
}

/** Range request for 2 bytes: cheap existence check that CDNs answer. */
function probe(url) {
  if (probes.has(url)) return probes.get(url);
  const result = (async () => {
    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), PROBE_TIMEOUT_MS);
    try {
      const res = await fetch(url, {
        method: "GET",
        headers: { Range: "bytes=0-1" },
        credentials: "omit",
        cache: "no-store",
        signal: controller.signal
      });
      if (res.body && typeof res.body.cancel === "function") res.body.cancel().catch(() => {});
      if (res.status === 200 || res.status === 206) return true;
      if (res.status === 403 || res.status === 404 || res.status === 410) return false;
      return null;
    } catch (_) {
      return null;
    } finally {
      clearTimeout(timer);
    }
  })();
  probes.set(url, result);
  return result;
}

/* -------------------------------------------------------------- playlists */

async function fetchPreMutePlaylist(playlistUrl, body) {
  const altUrl = playlistUrl.replace(MUTED_PLAYLIST_NAME, "");
  if (altUrl === playlistUrl) return null;
  try {
    const res = await fetch(altUrl, { credentials: "omit", cache: "no-store" });
    if (!res.ok) return null;
    const alt = await res.text();
    if (!alt.startsWith("#EXTM3U") || alt.includes("-muted")) return null;
    const count = (s) => (s.match(/^[^#\s].*$/gm) || []).length;
    if (count(alt) !== count(body)) return null;
    return alt;
  } catch (_) {
    return null;
  }
}

/** Record the rendition directories listed in a master playlist. */
function recordQualities(body, playlistUrl) {
  if (!body.includes("#EXT-X-STREAM-INF")) return false;
  const lines = body.split("\n");
  let expectUrl = false;
  let count = 0;
  for (const raw of lines) {
    const line = raw.trim();
    if (line.startsWith("#EXT-X-STREAM-INF")) {
      expectUrl = true;
      continue;
    }
    if (!expectUrl || !line || line.startsWith("#")) continue;
    expectUrl = false;
    let abs;
    try {
      abs = new URL(line, playlistUrl);
    } catch (_) {
      continue;
    }
    const parts = abs.pathname.split("/");
    if (parts.length < 3) continue;
    const key = abs.origin + parts.slice(0, -2).join("/");
    const quality = parts[parts.length - 2];
    const list = qualities.get(key) || [];
    if (!list.includes(quality)) list.push(quality);
    qualities.set(key, list);
    count++;
  }
  if (count) log(`master:${count} renditions`, playlistUrl);
  return count > 0;
}

function parseSegments(lines) {
  const segments = [];
  let elapsed = 0;
  let duration = 0;
  for (let i = 0; i < lines.length; i++) {
    const line = lines[i].trim();
    if (line.startsWith("#EXTINF:")) {
      duration = parseFloat(line.slice("#EXTINF:".length)) || 0;
      continue;
    }
    if (!line || line.startsWith("#")) continue;
    const match = MUTED_SEGMENT.exec(line);
    segments.push({
      index: i,
      path: line,
      unmutedPath: match ? match[1] + match[2] : null,
      start: elapsed,
      duration
    });
    elapsed += duration;
    duration = 0;
  }
  return segments;
}

function groupMutedRanges(segments) {
  const ranges = [];
  let current = null;
  for (const seg of segments) {
    if (!seg.unmutedPath) {
      current = null;
      continue;
    }
    if (!current) {
      current = { start: seg.start, end: seg.start + seg.duration, segments: [seg] };
      ranges.push(current);
    } else {
      current.end = seg.start + seg.duration;
      current.segments.push(seg);
    }
  }
  return ranges;
}

async function processPlaylist(originalBody, playlistUrl, tabId) {
  let body = originalBody;
  const report = reportFor(tabId);
  report.sawPlaylist = true;
  counters.playlists++;

  if (MUTED_PLAYLIST_NAME.test(playlistUrl)) {
    const preMute = await fetchPreMutePlaylist(playlistUrl, body);
    if (preMute) {
      report.note = "Served the pre-mute playlist for this highlight.";
      report.ranges = [];
      report.restoredRanges = 1;
      setBadge(tabId, report);
      log("playlist:pre-mute", playlistUrl);
      return preMute;
    }
  }

  if (recordQualities(body, playlistUrl)) return body; // master playlist

  if (!body.includes("-muted")) {
    log("playlist:clean", playlistUrl);
    if (report.ranges.length === 0) report.note = "Playlist has no muted segments.";
    setBadge(tabId, report);
    return body;
  }

  const lines = body.split("\n");
  const segments = parseSegments(lines);
  const ranges = groupMutedRanges(segments);
  if (ranges.length === 0) {
    log("playlist:clean", playlistUrl);
    setBadge(tabId, report);
    return body;
  }

  counters.muted += ranges.length;

  const muted = [];
  for (const range of ranges) for (const seg of range.segments) muted.push(seg);

  // Probe EVERY muted segment. Twitch purges the unmuted originals piecemeal,
  // so availability varies inside a single muted range: one probe per range
  // both skips recoverable audio and claims audio it can't actually deliver.
  const deadlineAt = Date.now() + PROBE_DEADLINE_MS;
  const budget = muted.slice(0, PROBE_BUDGET);
  for (const seg of budget) seg.unmutedUrl = new URL(seg.unmutedPath, playlistUrl).href;
  const primary = await probeMany(budget.map((s2) => s2.unmutedUrl), deadlineAt);
  for (const seg of budget) seg.available = primary.get(seg.unmutedUrl);
  for (const seg of muted.slice(PROBE_BUDGET)) {
    seg.unmutedUrl = new URL(seg.unmutedPath, playlistUrl).href;
    seg.available = null; // over budget: let the optimistic/fallback path decide
  }

  // Second chance: the same segment number under another quality rendition.
  if (settings.crossQuality) {
    const failed = muted.filter((seg) => seg.available === false);
    for (const seg of failed) {
      if (Date.now() > deadlineAt) break;
      for (const alt of altSegmentUrls(seg.unmutedUrl).slice(0, 3)) {
        if (await probe(alt)) {
          seg.altUrl = alt;
          seg.available = true;
          counters.crossQuality++;
          break;
        }
      }
    }
  }

  let restoredSegments = 0;
  for (const range of ranges) {
    range.ok = 0;
    range.guessed = 0;
    for (const seg of range.segments) {
      const rewrite = seg.available === true || (seg.available === null && settings.optimistic);
      if (!rewrite) continue;
      const target = seg.altUrl || seg.unmutedPath;
      lines[seg.index] = lines[seg.index].replace(seg.path, target);
      track(seg.altUrl || seg.unmutedUrl, new URL(seg.path, playlistUrl).href);
      segmentRange.set(seg.altUrl || seg.unmutedUrl, { tabId, key: vodKeyFor(playlistUrl), start: range.start });
      range.ok++;
      if (seg.available === null) range.guessed++;
      restoredSegments++;
    }
  }

  const airMatch = AIR_DATE.exec(body);
  const airedAt = airMatch ? Date.parse(airMatch[1].trim()) : NaN;

  report.note = null;
  report.key = vodKeyFor(playlistUrl);
  report.ranges = ranges.map((r) => ({
    id: r.start,
    start: formatTime(r.start),
    end: formatTime(r.end),
    seconds: Math.round(r.end - r.start),
    total: r.segments.length,
    ok: r.ok,
    failed: 0,
    guessed: r.guessed > 0,
    restored: r.ok > 0
  }));
  report.restoredRanges = report.ranges.filter((r) => r.ok === r.total).length;
  report.partialRanges = report.ranges.filter((r) => r.ok > 0 && r.ok < r.total).length;
  report.restoredSegments = restoredSegments;
  report.mutedSegments = muted.length;
  report.ageHours = Number.isNaN(airedAt) ? report.ageHours : (Date.now() - airedAt) / 3600000;
  setBadge(tabId, report);
  log(
    `playlist:${restoredSegments}/${muted.length} segments unmuted ` +
      `(${report.restoredRanges} full, ${report.partialRanges} partial)`,
    playlistUrl
  );

  return lines.join("\n");
}

/* --------------------------------------------- GraphQL (the red seek bar) */

function collectApiRanges(value, out) {
  if (Array.isArray(value)) {
    for (const item of value) collectApiRanges(item, out);
    return;
  }
  if (!value || typeof value !== "object") return;
  if (typeof value.offset === "number" && typeof value.duration === "number") {
    out.push({
      start: formatTime(value.offset),
      end: formatTime(value.offset + value.duration),
      seconds: value.duration
    });
    return;
  }
  for (const key of Object.keys(value)) collectApiRanges(value[key], out);
}

function stripMutedSegments(value, out) {
  if (Array.isArray(value)) {
    for (const item of value) stripMutedSegments(item, out);
    return;
  }
  if (!value || typeof value !== "object") return;
  for (const key of Object.keys(value)) {
    const child = value[key];
    if (MUTED_KEY.test(key) && child !== null && child !== undefined) {
      collectApiRanges(child, out);
      if (Array.isArray(child)) value[key] = [];
      else if (typeof child === "object" && "nodes" in child) value[key] = Object.assign({}, child, { nodes: [] });
      else value[key] = null;
      continue;
    }
    stripMutedSegments(child, out);
  }
}

function processGql(body, tabId) {
  if (!/muted/i.test(body)) return body;
  let data;
  try {
    data = JSON.parse(body);
  } catch (_) {
    return body;
  }
  const found = [];
  stripMutedSegments(data, found);
  if (found.length === 0) return body;

  const report = reportFor(tabId);
  counters.gql++;
  report.apiRanges = found;
  if (settings.hideMarkers) report.markersHidden += found.length;
  setBadge(tabId, report);
  log(`gql:${settings.hideMarkers ? "stripped" : "seen"} ${found.length} marker(s)`, "https://gql.twitch.tv/gql");

  return settings.hideMarkers ? JSON.stringify(data) : body;
}

/* --------------------------------------------------------------- plumbing */

function attachFilter(details, transform, { sniff }) {
  let filter;
  try {
    filter = browser.webRequest.filterResponseData(details.requestId);
  } catch (_) {
    return;
  }

  const chunks = [];
  let passthrough = false;

  filter.ondata = (event) => {
    if (sniff && chunks.length === 0 && !passthrough) {
      const head = new TextDecoder("utf-8", { fatal: false }).decode(event.data.slice(0, 8));
      if (!head.startsWith("#EXTM3U")) {
        passthrough = true;
        filter.write(event.data);
        filter.disconnect();
        return;
      }
      counters.sniffed++;
    }
    if (passthrough) {
      filter.write(event.data);
      return;
    }
    chunks.push(new Uint8Array(event.data));
  };

  filter.onerror = () => {};

  filter.onstop = async () => {
    if (passthrough) {
      filter.close();
      return;
    }
    const total = chunks.reduce((n, c) => n + c.length, 0);
    if (total === 0) {
      filter.close();
      return;
    }
    const raw = new Uint8Array(total);
    let offset = 0;
    for (const chunk of chunks) {
      raw.set(chunk, offset);
      offset += chunk.length;
    }
    try {
      const text = new TextDecoder("utf-8").decode(raw);
      const out = await transform(text, details);
      filter.write(out === text ? raw : new TextEncoder().encode(out));
    } catch (err) {
      filter.write(raw); // never break playback: pass the original through
    }
    filter.close();
  };
}

browser.webRequest.onBeforeRequest.addListener(
  (details) => {
    if (!settings.enabled) return;
    if (details.tabId < 0) return; // our own probe / alt-playlist fetches
    const url = details.url;
    if (!url.startsWith("http")) return;

    let host;
    try {
      host = new URL(url).host;
    } catch (_) {
      return;
    }

    if (IS_PLAYLIST.test(url)) {
      log("playlist:intercept", url);
      attachFilter(details, (text, d) => processPlaylist(text, d.url, d.tabId), { sniff: false });
      return;
    }

    if (host === "gql.twitch.tv" || (GQL_HOST.test(host) && /\/gql/.test(url))) {
      attachFilter(details, (text, d) => processGql(text, d.tabId), { sniff: false });
      return;
    }

    // Extension-less URL on a Twitch CDN: sniff for #EXTM3U, pass through if not.
    if (
      VOD_HOST.test(host) &&
      details.type === "xmlhttprequest" &&
      !IS_SEGMENT.test(url) &&
      !/\.(js|css|png|jpe?g|webp|gif|svg|woff2?|json)(\?|$)/i.test(url)
    ) {
      attachFilter(details, (text, d) => processPlaylist(text, d.url, d.tabId), { sniff: true });
      return;
    }

    if (VOD_HOST.test(host) && IS_SEGMENT.test(url) && /-muted\./.test(url)) {
      log("segment:muted-passed", url); // playlist rewrite missed this one
    }
  },
  { urls: ["<all_urls>"] },
  ["blocking"]
);

// If an unmuted segment we asked for is gone, quietly go back to the muted one.
browser.webRequest.onHeadersReceived.addListener(
  (details) => {
    if (details.statusCode < 400) return;
    const fallback = rewritten.get(details.url);
    if (!fallback) return;
    rewritten.delete(details.url);
    counters.fallbacks++;
    const report = tabReports.get(details.tabId);
    const where = segmentRange.get(details.url);
    if (report) {
      report.failedSegments++;
      if (where) {
        const range = report.ranges.find((r) => r.id === where.start);
        if (range) {
          range.failed++;
          range.ok = Math.max(0, range.ok - 1);
          range.restored = range.ok > 0;
        }
        report.restoredRanges = report.ranges.filter((r) => r.ok === r.total).length;
        report.partialRanges = report.ranges.filter((r) => r.ok > 0 && r.ok < r.total).length;
      }
      setBadge(details.tabId, report);
    }
    log("segment:fallback", details.url);
    return { redirectUrl: fallback };
  },
  { urls: ["<all_urls>"] },
  ["blocking", "responseHeaders"]
);

browser.tabs.onRemoved.addListener((tabId) => tabReports.delete(tabId));

browser.runtime.onMessage.addListener((message) => {
  if (message.type === "getReport") {
    return Promise.resolve({
      settings,
      report: tabReports.get(message.tabId) || null,
      diag: diag.slice(-25).reverse(),
      counters
    });
  }
  if (message.type === "setSetting") {
    return browser.storage.local.set({ [message.key]: message.value });
  }
  if (message.type === "reset") {
    tabReports.delete(message.tabId);
    probes.clear();
    rewritten.clear();
    segmentRange.clear();
    diag.length = 0;
    return Promise.resolve(true);
  }
  return false;
});
